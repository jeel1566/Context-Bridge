=== Issue 1: Logging Completely Non-Functional

*File:* [`backend/function_app.py`](backend/function_app.py:7)  
*Severity:* 🔴 CRITICAL  
*Type:* Observability

==== Description

The codebase imports `logging` module everywhere but has **no configuration**. This means:

* Logs go to stdout/stderr (Azure Functions default)
* No log levels are configured
* No structured logging (JSON format)
* No correlation IDs for request tracing
* No audit logging for sensitive operations
* No integration with Azure Application Insights

==== Impact

* Cannot debug production issues
* No visibility into system behavior
* Cannot trace requests across functions
* No audit trail for compliance
* Cannot monitor system health

==== Recommended Fix

Create `backend/logging_config.py`:

[source,python]
----
import logging
import json
import os
from datetime import datetime
from typing import Optional

class JsonFormatter(logging.Formatter):
    """Structured JSON formatter for production logging."""
    
    def format(self, record):
        log_data = {
            'timestamp': datetime.utcnow().isoformat(),
            'level': record.levelname,
            'logger': record.name,
            'message': record.getMessage(),
            'environment': os.environ.get('ENVIRONMENT', 'development'),
        }
        
        # Add correlation ID if available
        if hasattr(record, 'correlation_id'):
            log_data['correlation_id'] = record.correlation_id
        
        # Add user ID if available
        if hasattr(record, 'user_id'):
            log_data['user_id'] = record.user_id
        
        # Add request ID if available
        if hasattr(record, 'request_id'):
            log_data['request_id'] = record.request_id
        
        if record.exc_info:
            log_data['exception'] = self.formatException(record.exc_info)
        
        return json.dumps(log_data)

def configure_logging():
    """Configure structured logging for Context Bridge."""
    
    log_level = os.environ.get('LOG_LEVEL', 'INFO').upper()
    environment = os.environ.get('ENVIRONMENT', 'development')
    
    # Configure root logger
    root_logger = logging.getLogger()
    root_logger.setLevel(log_level)
    
    # Remove existing handlers
    root_logger.handlers.clear()
    
    # Console handler with JSON formatter
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(JsonFormatter())
    root_logger.addHandler(console_handler)
    
    # File handler for production
    if environment == 'production':
        from azure.monitor.opentelemetry import configure_azure_monitoring
        configure_azure_monitoring()
    
    return root_logger
----

from typing import Any, Dict, List, Optional
import re

class ValidationError(Exception):
    """Raised when validation fails."""
    pass

def validate_memory_data(data: Dict[str, Any]) -> Dict[str, Any]:
    """Validate memory data before storage."""
    
    # Validate title
    if 'title' not in data or not data['title']:
        raise ValidationError("Title is required")
    
    title = str(data['title']).strip()
    if len(title) > 200:
        raise ValidationError("Title must be less than 200 characters")
    
    # Validate content
    if 'content' not in data or not data['content']:
        raise ValidationError("Content is required")
    
    content = str(data['content']).strip()
    if len(content) > 10000:
        raise ValidationError("Content must be less than 10,000 characters")
    
    # Sanitize content (basic XSS prevention)
    content = sanitize_html(content)
    
    return {
        'title': title,
        'content': content,
        'tags': data.get('tags', [])
    }

def sanitize_html(text: str) -> str:
    """Basic HTML sanitization to prevent XSS."""
    # Remove dangerous tags
    dangerous_tags = ['script', 'iframe', 'object', 'embed', 'form']
    for tag in dangerous_tags:
        text = re.sub(f'<{tag}.*?>.*?</{tag}>', '', text, flags=re.IGNORECASE | re.DOTALL)
        text = re.sub(f'<{tag}.*?/>', '', text, flags=re.IGNORECASE)
    
    # Remove dangerous attributes
    dangerous_attrs = ['on\w+', 'javascript:', 'data:']
    for attr in dangerous_attrs:
        text = re.sub(f'{attr}=["\'][^"\']*["\']', '', text, flags=re.IGNORECASE)
    
    return text
----

# Add to backend/middleware/rate_limiter.py
import time
from collections import defaultdict
from typing import Dict
import logging

logger = logging.getLogger(__name__)

class RateLimiter:
    """Simple in-memory rate limiter."""
    
    def __init__(self, max_requests: int = 100, window_seconds: int = 60):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self.requests: Dict[str, List[float]] = defaultdict(list)
    
    def is_allowed(self, identifier: str) -> bool:
        """Check if request is allowed for identifier."""
        now = time.time()
        
        # Clean old requests
        self.requests[identifier] = [
            req_time for req_time in self.requests[identifier]
            if now - req_time < self.window_seconds
        ]
        
        # Check if under limit
        if len(self.requests[identifier]) >= self.max_requests:
            logger.warning(f"Rate limit exceeded for {identifier}")
            return False
        
        # Record request
        self.requests[identifier].append(now)
        return True
    
    def get_remaining(self, identifier: str) -> int:
        """Get remaining requests for identifier."""
        return self.max_requests - len(self.requests[identifier])

# Create rate limiters for different endpoints
memory_limiter = RateLimiter(max_requests=100, window_seconds=60)
auth_limiter = RateLimiter(max_requests=5, window_seconds=60)
share_limiter = RateLimiter(max_requests=10, window_seconds=60)

==== Impact

==== Recommended Fix

Add retry logic with exponential backoff:

[source,javascript]
----
async function refreshAccessToken(maxRetries = 3) {
    for (let attempt = 0; attempt < maxRetries; attempt++) {
        try {
            const response = await fetch(`${API_BASE_URL}/auth/refresh`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ refreshToken: result.refreshToken }),
            });
            
            if (response.ok) {
                const data = await response.json();
                await setAuthToken(data.data.access_token);
                return true;
            }
            
            // Exponential backoff
            if (attempt < maxRetries - 1) {
                await new Promise(resolve => setTimeout(resolve, Math.pow(2, attempt) * 1000));

from typing import Dict, Any, Type, TypeVar, Optional

T = TypeVar('T')

class ServiceContainer:
    """Simple dependency injection container."""
    
    def __init__(self):
        self._services: Dict[Type, Any] = {}
        self._singletons: Dict[Type, Any] = {}
    
    def register(self, interface: Type[T], implementation: Type[T], singleton: bool = True):
        """Register a service implementation."""
        if singleton:
            self._singletons[interface] = implementation
        else:
            self._services[interface] = implementation

        if interface in self._singletons:

                self._singletons[interface] = self._singletons[interface]()
            return self._singletons[interface]
        
        if interface in self._services:
            return self._services[interface]()
        
        raise ValueError(f"Service not registered: {interface}")

import time
from typing import Dict, Set
from collections import defaultdict
import logging

logger = logging.getLogger(__name__)

class IPBlocklist:
    """IP-based blocking service."""
    
    def __init__(self, max_attempts: int = 10, block_duration: int = 3600):
        self.max_attempts = max_attempts
        self.block_duration = block_duration  # 1 hour
        self.attempts: Dict[str, list] = defaultdict(list)
        self.blocked_ips: Dict[str, float] = {}
    
    def record_failed_attempt(self, ip: str) -> bool:
        """Record a failed attempt from IP."""
        now = time.time()
        
        # Check if already blocked
        if self.is_blocked(ip):
            return True
        
        # Clean old attempts
        self.attempts[ip] = [
            attempt for attempt in self.attempts[ip]
            if now - attempt < self.block_duration
        ]
        
        # Add new attempt
        self.attempts[ip].append(now)
        
        # Check if should block
        if len(self.attempts[ip]) >= self.max_attempts:
            self.blocked_ips[ip] = now + self.block_duration
            logger.warning(f"IP blocked: {ip}")
            return True
        
        return False
    
    def is_blocked(self, ip: str) -> bool:
        """Check if IP is blocked."""
        if ip not in self.blocked_ips:
            return False
        
        if time.time() > self.blocked_ips[ip]:
            # Block expired
            del self.blocked_ips[ip]
            return False
        
        return True
    
    def unblock(self, ip: str):
        """Unblock an IP."""
        if ip in self.blocked_ips:
            del self.blocked_ips[ip]
        if ip in self.attempts:
            del self.attempts[ip]

async def retry_with_backoff(
    func: Callable,
    max_retries: int = 3,
    base_delay: float = 1.0
) -> Optional[any]:
    """Retry function with exponential backoff."""
    for attempt in range(max_retries):

import time
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"

class CircuitBreaker:
    """Circuit breaker pattern for preventing cascading failures."""
    
    def __init__(self, failure_threshold: int = 5, timeout: int = 60):
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
    
    def record_failure(self):
        """Record a failure."""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
            logger.warning(f"Circuit breaker opened after {self.failure_count} failures")
    
    def record_success(self):
        """Record a success."""
        self.failure_count = 0
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.CLOSED
            logger.info("Circuit breaker closed")
    
    def can_execute(self) -> bool:
        """Check if execution is allowed."""
        if self.state == CircuitState.CLOSED:
            return True
        
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.timeout:
                self.state = CircuitState.HALF_OPEN
                logger.info("Circuit breaker half-open")
                return True
            return False
        
        return True

class AutoDiscoveryParser {
    constructor() {
        this.maxCandidates = 100;
        this.sufficientResults = 20; // Stop if we find 20 good candidates
        this.confidenceThreshold = 0.8;
    }
    
    async findTextCandidates() {
        const candidates = [];
        const startTime = Date.now();
        const timeout = 5000;
        
        const allElements = await this.querySelectorAllDeep('*', {
            timeout,
            maxElements: this.maxCandidates
        });
        
        for (const element of allElements) {
            // Check timeout
            if (Date.now() - startTime > timeout) {
                console.warn('Discovery timed out');
                break;
            }
            
            // Check if we have enough results
            if (candidates.length >= this.sufficientResults) {
                console.log(`Found sufficient results: ${candidates.length}`);
                break;
            }
            
            // Score the element
            const score = this.scoreElement(element);
            if (score >= this.confidenceThreshold) {
                candidates.push({
                    element,
                    score,
                    text: element.textContent
                });
            }
        }
        
        return candidates;
    }
    
    scoreElement(element) {
        """Score an element's likelihood of being a text candidate."""
        let score = 0;
        
        // Check for text content
        if (element.textContent && element.textContent.length > 50) {
            score += 0.3;
        }
        
        // Check for semantic tags
        if (['p', 'div', 'span', 'article'].includes(element.tagName.toLowerCase())) {
            score += 0.2;
        }
        
        // Check for visible
        if (element.offsetParent !== null) {
            score += 0.3;
        }
        
        // Check for class names
        if (element.className && /text|content|message/.test(element.className)) {
            score += 0.2;
        }
        
        return score;
    }
}
